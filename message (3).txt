local function New(class, props)
	local inst = Instance.new(class)
	local parent = props.Parent
	props.Parent = nil
	for k, v in pairs(props) do inst[k] = v end
	inst.Parent = parent
	return inst
end

local function Stroke(parent, color, transparency)
	return New("UIStroke", {
		Color = color or Theme.Outline,
		Transparency = transparency or 0,
		Thickness = 1,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
		Parent = parent,
	})
end

local function Label(props)
	props.BackgroundTransparency = 1
	props.BorderSizePixel = 0
	props.Font = props.Font or FONT
	props.TextSize = props.TextSize or 11
	props.TextColor3 = props.TextColor3 or Theme.Text
	props.TextXAlignment = props.TextXAlignment or Enum.TextXAlignment.Left
	return New("TextLabel", props)
end

local function BindClick(obj, callback)
	local busy = false
	local function fire(input)
		if busy then return end
		busy = true
		task.delay(0.08, function() busy = false end)
		pcall(callback, input)
	end
	pcall(function() obj.Activated:Connect(function(input) fire(input) end) end)
	pcall(function() obj.MouseButton1Click:Connect(function() fire(nil) end) end)
end

local function MakeDraggable(handle, target)
	local dragging, dragStart, startPos = false, nil, nil
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1
			or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = target.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then dragging = false end
			end)
		end
	end)
	UIS.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch) then
			local d = input.Position - dragStart
			target.Position = UDim2.new(
				startPos.X.Scale, startPos.X.Offset + d.X,
				startPos.Y.Scale, startPos.Y.Offset + d.Y
			)
		end
	end)
end

local function Round(v, decimals)
	local m = 10 ^ (decimals or 0)
	return mathfloor(v * m + 0.5) / m
end

local order = 0
local function NextOrder() order += 1; return order end

local Library = {}
Library.ConfigStore = {}

function Library:RegisterSaver(key, getter, setter)
	Library.ConfigStore[key] = { get = getter, set = setter }
end

function Library:SerializeConfig()
	local out = {}
	for k, v in pairs(Library.ConfigStore) do
		local ok, val = pcall(v.get)
		if ok then out[k] = val end
	end
	return out
end

function Library:ApplyConfig(data)
	if type(data) ~= "table" then return end
	for k, v in pairs(data) do
		local entry = Library.ConfigStore[k]
		if entry and entry.set then
			pcall(entry.set, v)
		end
	end
end

			function Section:AddSlider(o)
				local Slider = { Value = o.Default or o.Min, _callback = o.Callback }
				local dec = o.Decimals or 0
				local holder = New("Frame", {
					Size = UDim2.new(1, 0, 0, 31), BackgroundTransparency = 1,
					LayoutOrder = NextOrder(), Parent = body,

					Position = UDim2.fromOffset(2, 16), Size = UDim2.new(1, -4, 0, 12),

				})
				Stroke(bar, Theme.Outline)
				local fill = New("Frame", {
					Size = UDim2.fromScale(0, 1), BackgroundColor3 = Theme.Accent,
					BorderSizePixel = 0, Parent = bar,
				})
				local valLbl = Label({ Size = UDim2.fromScale(1, 1),
					TextXAlignment = Enum.TextXAlignment.Center, TextSize = 10, Text = "", ZIndex = 2, Parent = bar })

				local function fmt()
					if o.Format then return o.Format(Slider.Value, o.Max) end
					return ("%s/%s"):format(tostring(Slider.Value), tostring(o.Max))
				end

				function Slider:Set(v)
					v = Round(mathclamp(v, o.Min, o.Max), dec)
					self.Value = v
					fill.Size = UDim2.fromScale((v - o.Min) / (o.Max - o.Min), 1)
					valLbl.Text = fmt()
					if self._callback then task.spawn(self._callback, v) end
				end
				function Slider:Get() return self.Value end

				local sliding = false
				local function update(x)
					local rel = mathclamp((x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
					Slider:Set(o.Min + (o.Max - o.Min) * rel)
				end
				bar.InputBegan:Connect(function(i)
					if i.UserInputType == Enum.UserInputType.MouseButton1
						or i.UserInputType == Enum.UserInputType.Touch then
						sliding = true; update(i.Position.X)
					end
				end)
				table.insert(Window.Connections, UIS.InputEnded:Connect(function(i)
					if i.UserInputType == Enum.UserInputType.MouseButton1
						or i.UserInputType == Enum.UserInputType.Touch then sliding = false end
				end))
				table.insert(Window.Connections, UIS.InputChanged:Connect(function(i)
					if sliding and (i.UserInputType == Enum.UserInputType.MouseMovement
						or i.UserInputType == Enum.UserInputType.Touch) then update(i.Position.X) end
				end))

				if o.ConfigKey then
					Library:RegisterSaver(o.ConfigKey,
						function() return Slider.Value end,
						function(v) if v ~= Slider.Value then Slider:Set(v) end end)
				end

				Slider:Set(Slider.Value)
				return Slider
			end

cheat.utility.get_multipoint = LPH_JIT_MAX(function(model, aimpart)
	if not model then return nil end
	local part = _FindFirstChild(model, aimpart or "Head") or _FindFirstChild(model, "Head")
	if not (part and part:IsA("BasePart")) then return nil end
	local hs = modulecomm and modulecomm.hitbox_size
	local sx, sy, sz = part.Size.X, part.Size.Y, part.Size.Z
	if hs and type(hs.X) == "number" and hs.X > 0 and hs.X <= 1000
	   and type(hs.Y) == "number" and hs.Y > 0 and hs.Y <= 1000 then
		sy = hs.Y
		sz = type(hs.Z) == "number" and hs.Z > 0 and hs.Z <= 1000 and hs.Z or hs.X
		sx = hs.X
	end
	local hx, hy, hz = sx * 0.5, sy * 0.5, sz * 0.5
	local rot = part.CFrame - part.CFrame.Position
	local camPos = Camera.CFrame.Position
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	local filt = { model }
	local char = LocalPlayer.Character
	if char then filt[#filt + 1] = char end
	local ignore = Workspace:FindFirstChild("Const") and Workspace.Const:FindFirstChild("Ignore")
	if ignore then filt[#filt + 1] = ignore end
	params.FilterDescendantsInstances = filt
	local function isVis(p)
		local dir = p - camPos
		if dir.Magnitude < 1 then return true end
		local r = _Raycast(Workspace, camPos, dir, params)
		return (not r) or (r.Instance and _IsDescendantOf(r.Instance, model))
	end
	local center = part.Position
	if isVis(center) then return center end
	local points = {
		_Vector3new( hx,  hy,  hz), _Vector3new(-hx,  hy,  hz),
		_Vector3new( hx,  hy, -hz), _Vector3new(-hx,  hy, -hz),
		_Vector3new(  0,  hy,  hz), _Vector3new( hx,  hy,   0),
		_Vector3new(-hx,  hy,   0), _Vector3new(  0,  hy, -hz),
		_Vector3new( hx,   0,  hz), _Vector3new(-hx,   0,  hz),
		_Vector3new( hx,   0, -hz), _Vector3new(-hx,   0, -hz),
		_Vector3new(  0,   0,  hz), _Vector3new(  0,   0, -hz),
		_Vector3new( hx,   0,   0), _Vector3new(-hx,   0,   0),
		_Vector3new( hx, -hy,  hz), _Vector3new(-hx, -hy,  hz),
		_Vector3new( hx, -hy, -hz), _Vector3new(-hx, -hy, -hz),
	}
	local best, bestd = nil, math.huge
	for i = 1, #points do
		local p = center + (rot * points[i])
		if isVis(p) then
			local d = (p - center).Magnitude
			if d < bestd then bestd = d; best = p end
		end
	end
	return best
end)

local SpeedCfg = {
	bhop_enabled = false, bhop_silent = false, speed = 55,
	downcliff_mode = false, downcliff_speed = 150, downcliff_accel = 50, downcliff_fall = 50, downcliff_up = 50,
	jetpackabuse = false, abuseupspeed = 55, abusedownspeed = 0,
	forcesprint = false, tp_up_studs = 6,
}
do
	local middle = Workspace.Const.Ignore.LocalCharacter.Middle
	local bottom = Workspace.Const.Ignore.LocalCharacter.Bottom
	local top    = Workspace.Const.Ignore.LocalCharacter.Top
	local characterparams = RaycastParams.new()
	characterparams.IgnoreWater = true
	characterparams.FilterDescendantsInstances = { Workspace.Const.Ignore }
	characterparams.FilterType = Enum.RaycastFilterType.Exclude
	characterparams.RespectCanCollide = true

	local niga, wtf = SpeedCfg.speed, 0
	local dc_buildup, dc_oldpos = SpeedCfg.downcliff_speed, middle.Position

	cheat.utility.new_renderstepped(LPH_JIT_MAX(function(delta)
		local shiftpressed = _IsKeyDown(UIS, Enum.KeyCode.LeftShift)
		local cpressed = _IsKeyDown(UIS, Enum.KeyCode.C)
		local spacedown = _IsKeyDown(UIS, Enum.KeyCode.Space)
		local cameralook = Camera.CFrame.LookVector
		local direction
		if middle and (cpressed or spacedown or shiftpressed or SpeedCfg.forcesprint or SpeedCfg.jetpackabuse) then
			local neutralized_look = _Vector3new(cameralook.X, 0, cameralook.Z).Unit
			direction = Vector3.zero
			direction = _IsKeyDown(UIS, Enum.KeyCode.W) and direction + neutralized_look or direction
			direction = _IsKeyDown(UIS, Enum.KeyCode.S) and direction - neutralized_look or direction
			direction = _IsKeyDown(UIS, Enum.KeyCode.D) and direction + _Vector3new(-neutralized_look.Z, 0, neutralized_look.X) or direction
			direction = _IsKeyDown(UIS, Enum.KeyCode.A) and direction + _Vector3new(neutralized_look.Z, 0, -neutralized_look.X) or direction
		end
		if SpeedCfg.bhop_silent then return end
		wait()
		if SpeedCfg.bhop_enabled and not SpeedCfg.downcliff_mode and middle and cpressed and shiftpressed then
			if not (direction == Vector3.zero) then direction = direction.Unit end
			niga = mathclamp(niga - delta * 20, 17, SpeedCfg.speed)
			if wtf == 0 then
				local tpcf, mdcf, bmcf = top.CFrame, middle.CFrame, bottom.CFrame
				middle.CFrame = mdcf + _Vector3new(0, SpeedCfg.tp_up_studs, 0)
				bottom.CFrame = bmcf + _Vector3new(0, SpeedCfg.tp_up_studs, 0)
				top.CFrame = tpcf + _Vector3new(0, SpeedCfg.tp_up_studs, 0)
			end
			middle.AssemblyLinearVelocity = (direction * niga) + (Vector3.yAxis * (wtf < 0.85 and 0 or -7))
			bottom.AssemblyLinearVelocity = (direction * niga) + (Vector3.yAxis * (wtf < 0.85 and 0 or -7))
			top.AssemblyLinearVelocity = (direction * niga) + (Vector3.yAxis * (wtf < 0.85 and 0 or -7))
			wtf = wtf + delta
		elseif SpeedCfg.bhop_enabled and SpeedCfg.downcliff_mode and middle and cpressed and shiftpressed then
			if not (direction == Vector3.zero) then direction = direction.Unit end
			local wattasigma = middle.Position
			local speed_dir = direction * dc_buildup * delta
			local verticalrc = _Raycast(Workspace, wattasigma, -Vector3.yAxis * 20, characterparams)
			local horizontalrc = _Raycast(Workspace, verticalrc and verticalrc.Position or wattasigma, speed_dir, characterparams)
			dc_buildup = mathclamp(verticalrc and dc_buildup + SpeedCfg.downcliff_accel * math.abs(wattasigma.Y - verticalrc.Position.Y) or dc_buildup - delta * 30, 17, 250)
			if horizontalrc or verticalrc then
				local middle_pos = (horizontalrc or verticalrc).Position + Vector3.yAxis * 1.5
				local bottom_offset = bottom.Position - middle_pos
				local top_offset = top.Position - middle_pos
				middle.CFrame = _CFramenew(middle_pos)
				bottom.CFrame = _CFramenew(middle_pos + bottom_offset)
				top.CFrame = _CFramenew(middle_pos + top_offset)
				middle.AssemblyLinearVelocity = Vector3.zero
				bottom.AssemblyLinearVelocity = Vector3.zero
				top.AssemblyLinearVelocity = Vector3.zero
			else
				middle.AssemblyLinearVelocity = (direction * dc_buildup) + (Vector3.yAxis * -7)
				bottom.AssemblyLinearVelocity = (direction * dc_buildup) + (Vector3.yAxis * -7)
				top.AssemblyLinearVelocity = (direction * dc_buildup) + (Vector3.yAxis * -7)
			end
		elseif SpeedCfg.jetpackabuse and middle then
			if not (direction == Vector3.zero) then direction = direction.Unit end
			local yvelo = spacedown and SpeedCfg.abuseupspeed or shiftpressed and -SpeedCfg.abusedownspeed or 0
			middle.AssemblyLinearVelocity = _Vector3new(direction.X * 18, yvelo, direction.Z * 18)
			top.AssemblyLinearVelocity = _Vector3new(direction.X * 18, yvelo, direction.Z * 18)
			bottom.AssemblyLinearVelocity = _Vector3new(direction.X * 18, yvelo, direction.Z * 18)
		else
			if SpeedCfg.forcesprint and middle then
				if not (direction == Vector3.zero) then direction = direction.Unit end
				middle.AssemblyLinearVelocity = _Vector3new(direction.X * 18, middle.AssemblyLinearVelocity.Y, direction.Z * 18)
			end
			niga = SpeedCfg.speed
			dc_buildup = 55
			dc_oldpos = middle.Position
			wtf = 0
		end
	end))
end

	local function CalculateBallistics(targetPos, targetVel, camPos, projSpeed, projDrop)
		targetPos = targetPos or camPos; targetVel = targetVel or Vector3.zero
		local distance = (targetPos - camPos).Magnitude
		local timeToTarget = distance / projSpeed
		local adjustedHorizontal = Vector3.new(targetVel.X, 0, targetVel.Z) * 7
		local adjustedVertical = Vector3.new(0, targetVel.Y, 0) * 2
		local predictedWorldPos = targetPos + ((adjustedHorizontal + adjustedVertical) * timeToTarget)
		local verticalDrop = -projDrop ^ (timeToTarget * projDrop) + 1
		return predictedWorldPos - Vector3.new(0, verticalDrop, 0), timeToTarget
	end

	local sa_sleep_cache = setmetatable({}, {__mode = "k"})
	local function sa_is_sleeper(model)
		if not model then return false end
		local cached = sa_sleep_cache[model]
		if cached and tick() - cached.t < 1 then return cached.v end
		local lt = _FindFirstChild(model, "LowerTorso")
		local result = false
		if lt then
			local rr = _FindFirstChild(lt, "RootRig")
			if rr then
				local ok, angle = pcall(function() return rr.CurrentAngle end)
				result = ok and type(angle) == "number" and angle ~= 0 or false
			end
		end
		sa_sleep_cache[model] = { v = result, t = tick() }
		return result
	end
	local function sa_is_team(model)
		if not model then return false end
		local h = _FindFirstChild(model, "Head")
		if h then
			local d = _FindFirstChild(h, "Dot")
			if d and d.Enabled then return true end
		end
		return false
	end
	local function sa_is_ai(model)
		local t = _FindFirstChild(model, "Torso") or _FindFirstChild(model, "HumanoidRootPart")
		return t and t.CollisionGroup == "NPC" or false
	end

	raycastParams.IgnoreWater = true
	local function IsTargetVisible(model)
		if not model then return false end
		local head = _FindFirstChild(model, "Head")
		if not head then return false end
		local filter = { model }
		if LocalPlayer.Character then table.insert(filter, LocalPlayer.Character) end
		raycastParams.FilterDescendantsInstances = filter
		local origin = Camera.CFrame.Position
		local direction = (head.Position - origin)
		local result = Workspace:Raycast(origin, direction, raycastParams)
		if not result then return true end
		if result.Instance and result.Instance:IsDescendantOf(model) then return true end
		return false
	end

	local function GetOptimalTarget()
		if not EntityMap then return nil, nil end
		local bestTarget, bestVelocity, minMagnitude = nil, nil, SilentAim.Settings.fovSize
		local viewportCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
		for entityIndex, entityData in next, EntityMap do
			local model = entityData.model
			if model and _FindFirstChild(model, "HumanoidRootPart") then
				local skip = false
				if SilentAim.Settings.sleeperCheck and sa_is_sleeper(model) then skip = true end
				if SilentAim.Settings.aiCheck and sa_is_ai(model) then skip = true end
				if SilentAim.Settings.teamCheck and sa_is_team(model) then skip = true end
				if not skip then
					local head = _FindFirstChild(model, "Head")
					if head then
						local screenPosition, isOnScreen = Camera:WorldToViewportPoint(head.Position)
						if isOnScreen then
							local mouseDistance = (Vector2.new(screenPosition.X, screenPosition.Y) - viewportCenter).Magnitude
							if mouseDistance < minMagnitude then
								minMagnitude = mouseDistance
								bestTarget = model
								bestVelocity = entityData.velocityVector or Vector3.zero
							end
						end
					end
				end
			end
		end
		return bestTarget, bestVelocity
	end

						end
					end

			end

		end

							VisibleText.Position = Vector2.new(pos.X + 10, pos.Y - 20)
							VisibleText.Visible = true
						else VisibleText.Visible = false end
					else SnapLine.Visible = false; VisibleText.Visible = false end
				else SnapLine.Visible = false; VisibleText.Visible = false end

	Font = Drawing.Fonts.System, rainbow = false, sussy = false,

	Callback = function(first) cursor.Dot = first end })
crossSec:AddToggle({ Name = "Sussy", Default = false, ConfigKey = "cross_sussy",
	Callback = function(first) cursor.sussy = first end })
crossSec:AddToggle({ Name = "Rainbow", Default = false, ConfigKey = "cross_rainbow",
	Callback = function(first) cursor.rainbow = first end })

do
	local lines = {}
	local outline = cheat.utility.new_drawing("Square", { Visible = true, Size = _Vector2new(4, 4), Color = BLACK, Filled = true, ZIndex = 1, Transparency = 1 })
	local dot = cheat.utility.new_drawing("Square", { Visible = true, Size = _Vector2new(2, 2), Color = cursor.Color, Filled = true, ZIndex = 2, Transparency = 1 })
	local logotext = cheat.utility.new_drawing("Text", { Visible = false, Font = cursor.Font, Size = 13, Color = PURPLE_LIGHT, ZIndex = 3, Transparency = 1, Text = "floppa hub", Center = true, Outline = true })
	for i = 1, 4 do
		local line_outline = cheat.utility.new_drawing("Line", { Visible = true, From = _Vector2new(200, 500), To = _Vector2new(200, 500), Color = BLACK, Thickness = cursor.Thickness + 2.5, ZIndex = 1, Transparency = 1 })
		local line = cheat.utility.new_drawing("Line", { Visible = true, From = _Vector2new(200, 500), To = _Vector2new(200, 500), Color = cursor.Color, Thickness = cursor.Thickness, ZIndex = 2, Transparency = 1 })
		local naziline = cheat.utility.new_drawing("Line", { Visible = true, From = _Vector2new(200, 500), To = _Vector2new(200, 500), Color = cursor.Color, Thickness = cursor.Thickness, ZIndex = 2, Transparency = 1 })
		lines[i] = { line, line_outline, naziline }
	end
	local angle, transp, reverse = 0, 0, false
	local pos, rainbow, rotationdegree, color = Vector2.zero, 0, 0, Color3.new()
	local math_cos, math_atan, math_pi, math_sin = math.cos, math.atan, math.pi, math.sin
	local function DEG2RAD(x) return x * math_pi / 180 end
	local function RAD2DEG(x) return x * 180 / math_pi end
	cheat.utility.new_renderstepped(LPH_NO_VIRTUALIZE(function(delta)
		if cursor.Enabled then
			rainbow = rainbow + (delta * 0.5)
			if rainbow > 1.0 then rainbow = 0.0 end
			pos = _Vector2new(Mouse.X, Mouse.Y + GuiInset.Y)
			if cursor.rainbow then color = Color3.fromHSV(rainbow, 1, 1) else color = cursor.Color end
			if transp <= 1.5 + cursor.Text.LogoFadingOffset and not reverse then
				transp = transp + (((cursor.Speed == 0 and 1 or cursor.Speed) * 10) * delta)
				if transp >= 1.5 + cursor.Text.LogoFadingOffset then reverse = true end
			elseif reverse then
				transp = transp - (((cursor.Speed == 0 and 1 or cursor.Speed) * 10) * delta)
				if transp <= 0 - cursor.Text.LogoFadingOffset then reverse = false end
			end
			logotext.Position = _Vector2new(pos.X, (pos + _Vector2new(0, cursor.Radius + 5)).Y)
			logotext.Transparency = transp
			logotext.Visible = cursor.Text.Logo
			logotext.Color = cursor.Text.LogoColor
			logotext.Font = cursor.Font
			if cursor.sussy then
				local frametime = delta
				local a = cursor.Radius - 10
				local gamma = math_atan(a / a)
				if rotationdegree >= 90 then rotationdegree = 0 end
				for i = 1, 4 do
					local p_0 = (a * math_sin(DEG2RAD(rotationdegree + (i * 90))))
					local p_1 = (a * math_cos(DEG2RAD(rotationdegree + (i * 90))))
					local p_2 = ((a / math_cos(gamma)) * math_sin(DEG2RAD(rotationdegree + (i * 90) + RAD2DEG(gamma))))
					local p_3 = ((a / math_cos(gamma)) * math_cos(DEG2RAD(rotationdegree + (i * 90) + RAD2DEG(gamma))))
					lines[i][1].From = _Vector2new(pos.X, pos.Y)
					lines[i][1].To = _Vector2new(pos.X + p_0, pos.Y - p_1)
					lines[i][1].Color = color; lines[i][1].Thickness = cursor.Thickness; lines[i][1].Visible = true
					lines[i][3].From = _Vector2new(pos.X + p_0, pos.Y - p_1)
					lines[i][3].To = _Vector2new(pos.X + p_2, pos.Y - p_3)
					lines[i][3].Color = color; lines[i][3].Thickness = cursor.Thickness; lines[i][3].Visible = true
				end
				rotationdegree = rotationdegree + ((cursor.Speed * frametime) * 1000)
			else
				angle = (cursor.Speed == 0 and 0 or angle + ((cursor.Speed * 10) * delta))
				if angle >= 90 then angle = 0 end
				dot.Visible = cursor.Dot; dot.Color = color
				dot.Position = _Vector2new(pos.X - 1, pos.Y - 1)
				outline.Visible = cursor.Outline and cursor.Dot
				outline.Position = _Vector2new(pos.X - 2, pos.Y - 2)
				for index, line in pairs(lines) do
					local x, y = {}, {}
					if cursor.Resize then
						x = { pos.X + (math_cos(angle + (index * (math_pi / 2))) * (cursor.Radius + ((cursor.Radius * math_sin(angle)) / 9))),
						      pos.X + (math_cos(angle + (index * (math_pi / 2))) * ((cursor.Radius - 20) - (cursor.TheGap and (((cursor.Radius - 20) * math_cos(angle)) / 4) or (((cursor.Radius - 20) * math_cos(angle)) - 4)))) }
						y = { pos.Y + (math_sin(angle + (index * (math_pi / 2))) * (cursor.Radius + ((cursor.Radius * math_sin(angle)) / 9))),
						      pos.Y + (math_sin(angle + (index * (math_pi / 2))) * ((cursor.Radius - 20) - (cursor.TheGap and (((cursor.Radius - 20) * math_cos(angle)) / 4) or (((cursor.Radius - 20) * math_cos(angle)) - 4)))) }
					else
						x = { pos.X + (math_cos(angle + (index * (math_pi / 2))) * (cursor.Radius)),
						      pos.X + (math_cos(angle + (index * (math_pi / 2))) * ((cursor.Radius - 20) - (cursor.TheGap and ((cursor.Radius - 20) / cursor.Gap) or ((cursor.Radius - 20) - cursor.Gap)))) }
						y = { pos.Y + (math_sin(angle + (index * (math_pi / 2))) * (cursor.Radius)),
						      pos.Y + (math_sin(angle + (index * (math_pi / 2))) * ((cursor.Radius - 20) - (cursor.TheGap and ((cursor.Radius - 20) / cursor.Gap) or ((cursor.Radius - 20) - cursor.Gap)))) }
					end
					line[1].Visible = true; line[1].Color = color
					line[1].From = _Vector2new(x[2], y[2]); line[1].To = _Vector2new(x[1], y[1])
					line[1].Thickness = cursor.Thickness
					line[2].Visible = cursor.Outline
					line[2].From = _Vector2new(x[2], y[2]); line[2].To = _Vector2new(x[1], y[1])
					line[2].Thickness = cursor.Thickness + 2.5
					line[3].Visible = false
				end
			end
		else
			dot.Visible = false; outline.Visible = false; logotext.Visible = false
			for _, line in pairs(lines) do line[1].Visible = false; line[2].Visible = false; line[3].Visible = false end
		end
	end))
end

	local middle = Workspace.Const.Ignore.LocalCharacter.Middle
	local bottom = Workspace.Const.Ignore.LocalCharacter.Bottom
	local top    = Workspace.Const.Ignore.LocalCharacter.Top
	local mdpos, bmpos, tppos = middle.CFrame, bottom.CFrame, top.CFrame

	fcSec:AddToggle({ Name = "Freecam Enabled", Default = false, Keybind = true, ConfigKey = "fc_enabled",
		Callback = function(first)

			top.CanCollide = not first
		end })
	fcSec:AddSlider({ Name = "Speed", Min = 1, Max = 150, Default = 10, ConfigKey = "fc_speed",
		Callback = function(State) speed = State end })
	fcSec:AddDropdown({ Name = "Part", Options = { "middle", "bottom", "top" }, Default = "middle", ConfigKey = "fc_part",
		Callback = function(Value) part = Value end })
	cheat.utility.new_heartbeat(LPH_JIT_MAX(function(delta)
		if enabled and middle then
			if part == "middle" then middle.CFrame = mdpos end
			if part == "bottom" then bottom.CFrame = bmpos end
			if part == "top" then top.CFrame = tppos end
			RunService.RenderStepped:Wait()
			if middle then
				local cameralook = Camera.CFrame.LookVector
				local direction = Vector3.zero
				direction = _IsKeyDown(UIS, Enum.KeyCode.W) and direction + cameralook or direction
				direction = _IsKeyDown(UIS, Enum.KeyCode.S) and direction - cameralook or direction
				direction = _IsKeyDown(UIS, Enum.KeyCode.D) and direction + _Vector3new(-cameralook.Z, 0, cameralook.X) or direction
				direction = _IsKeyDown(UIS, Enum.KeyCode.A) and direction + _Vector3new(cameralook.Z, 0, -cameralook.X) or direction
				if direction ~= Vector3.zero then direction = direction.Unit end
				freecamoffset = freecamoffset + (direction * delta * speed)
				if part == "middle" then middle.CFrame = mdpos + freecamoffset end
				if part == "bottom" then bottom.CFrame = bmpos + freecamoffset end
				if part == "top" then top.CFrame = tppos + freecamoffset end
				middle.AssemblyLinearVelocity = Vector3.zero
				bottom.AssemblyLinearVelocity = Vector3.zero
				top.AssemblyLinearVelocity = Vector3.zero
			end
		elseif middle then
			freecamoffset = Vector3.zero
			mdpos, bmpos, tppos = middle.CFrame, bottom.CFrame, top.CFrame
		end
	end))
end

	local function getConfigFiles()
		local files = {}
		if not canSave then return files end
		local ok, list = pcall(function()
			return listfiles(CONFIG_DIR)
		end)
		if ok and list then
			for _, f in ipairs(list) do
				local name = tostring(f):match("([^/\\]+)" .. CONFIG_EXT .. "$")
				if name then table.insert(files, name) end
			end
		end
		table.sort(files)
		return files
	end

	local function saveConfig(name)
		if not canSave then Notify("Config", "Executor doesn't support file write"); return end
		if typeof(isfolder) == "function" and typeof(makefolder) == "function" and not isfolder(CONFIG_DIR) then
			pcall(makefolder, CONFIG_DIR)
		end
		local data = Library:SerializeConfig()
		local ok, encoded = pcall(function() return HttpService:JSONEncode(data) end)
		if not ok then Notify("Config", "Encode failed"); return end
		local path = CONFIG_DIR .. "/" .. name .. CONFIG_EXT
		local wok = pcall(function() writefile(path, encoded) end)
		if wok then
			Notify("Config", "Saved: " .. name)
			currentConfigName = name
		else
			Notify("Config", "Write failed")
		end
	end

	local function loadConfig(name)
		if not canSave then Notify("Config", "Executor doesn't support file read"); return end
		local path = CONFIG_DIR .. "/" .. name .. CONFIG_EXT
		if typeof(isfile) == "function" and not isfile(path) then
			Notify("Config", "Not found: " .. name); return
		end
		local ok, content = pcall(function() return readfile(path) end)
		if not ok or not content then Notify("Config", "Read failed"); return end
		local dok, data = pcall(function() return HttpService:JSONDecode(content) end)
		if not dok or type(data) ~= "table" then Notify("Config", "Decode failed"); return end
		Library:ApplyConfig(data)
		currentConfigName = name
		Notify("Config", "Loaded: " .. name)
	end

					tripmode == "random" and math.sin(tick() * aaset.cartriprandomspeed) * math.rad(70) or
					tripmode == "pizdec" and math.sin(tick() * 360) * math.rad(80) or args[4].Y or 0)
				roll = (tripmode == "none" and args[4].Z or tripmode == "zero" and 0 or
					tripmode == "random" and math.cos(tick() * aaset.cartriprandomspeed) * math.rad(70) or
					tripmode == "pizdec" and math.cos(tick() * 360) * math.rad(80) or args[4].Z or 0)
				local yawset = aaset.antiaimyaw
				local yaw = ((yawset == "static" or yawset == "jitter") and
					(args[3].X + (math.rad(aaset.yawoffset) - (aaset.backwards and math.pi or 0)))) or 0
				yaw = (yawset == "none" and args[3].Y or
					yawset == "spin" and (tick() % 1) * ((aaset.spinspeed / 360) * math.pi) or
					yawset == "invisible" and math.huge or
					yawset == "jitter" and math.rad((real_extended % 2 == 0 and 1 or -1) * (aaset.yawjittersize / 2)) or yaw or 0)
				args[4] = _Vector3new(pitch, yaw, roll)
			end
		end
		if args[1] == 1 then
			real_extended = real_extended + 1
			if modulecomm.velobreak then
				if real_extended % 4 == 0 or real_extended % 4 == 2 then args[5] = nil end
				args[2] = args[2] + (real_extended % 4 == 1 and Vector3.zero or Vector3.yAxis * modulecomm.velobreakstrenght)
				return __gamenamecall(self, unpack(args))
			end
			if modulecomm.invis then args[5] = nil; return __gamenamecall(self, unpack(args)) end
			if modulecomm.invis2 then args[5] = nil; return __gamenamecall(self, unpack(args)) end
			trident.lastpos = args[2] + _Vector3new(0, 3, 0)
			if typeof(args[3]) == "Vector3" and modulecomm.antiaim.antiaim then
				local aaset = modulecomm.antiaim
				if aaset.antiaimyaw == "invisible" then
					args[3] = Vector3int16.new(0, math.huge, 0)
					return __gamenamecall(self, unpack(args))
				elseif aaset.antiaimyaw == "invisible2" then
					args[3] = Vector3int16.new(math.huge, 0, 0)
					return __gamenamecall(self, unpack(args))
				end
				local pitchset = aaset.antiaimpitch
				local pitch = (pitchset == "none" and args[3].Y or pitchset == "down" and -1.5 or pitchset == "up" and 1.5 or
					pitchset == "jitter" and (real_extended % 2 == 0 and -1.5 or 1.5) * (aaset.pitchjittersize / 90) or
					pitchset == "random" and math.sin(tick() * aaset.pitchrandomspeed) * math.rad(90) or
					pitchset == "custom" and 1.5 * (aaset.pitchcustom / 90) or args[3].Y or 0)
				local yawset = aaset.antiaimyaw
				local yaw = ((yawset == "static" or yawset == "jitter") and
					(args[3].X + math.rad(aaset.yawoffset + (aaset.backwards and 180 or 0)))) or 0
				yaw = (yawset == "none" and args[3].Y or
					yawset == "spin" and ((args[3].X - (aaset.backwards and math.pi or 0) + math.rad(aaset.yawoffset)) + ((tick() * ((aaset.spinspeed / 60) * (2 * math.pi))) % (2 * math.pi))) or
					yawset == "invisible" and math.huge or
					yawset == "random" and math.sin(tick() * aaset.yawrandomspeed) * math.pi or
					yawset == "jitter" and math.rad((real_extended % 2 == 0 and 1 or -1) * aaset.yawjittersize) or
					yawset == "custom" and math.rad(aaset.yawcustom + 360 + args[3].X - (aaset.backwards and 180 or 0)) or yaw or 0)
				args[3] = _Vector3new(pitch, yaw, 0)
			end
		end
		if args[1] == 1 and modulecomm.godmode then
			args[2] = Vector3.new(args[2].X, args[2].Y + (modulecomm.godmode_strenght or 6), args[2].Z)
		end
		if args[1] == 1 and modulecomm.forseminoclip then
			args[2] = Vector3.new(args[2].X, args[2].Y + 1.5, args[2].Z)
		end
		if args[1] == 2 and modulecomm.silentwalk then
			return __gamenamecall(self, 2, true)
		end
		if args[1] == 10 then
			if args[2] == "Hit" and type(args[5]) == "number" and type(args[6]) == "string" and typeof(args[7]) == "Vector3" then
				local entity = rawget(getupvalue(garbage.entitylist, 1), args[5])
				local entitymodel = entity and rawget(entity, "model")
				local entityhead = entitymodel and _FindFirstChild(entitymodel, "Head")
				if entityhead then
					if modulecomm.forcehead then args[6] = "Head" end
					local part = _FindFirstChild(entitymodel, args[6])
					local originalpart = _FindFirstChild(trident.original_model, args[6])
					if not (part and originalpart) then return __gamenamecall(self, unpack(args)) end
					if modulecomm.hitboxes then
						local primarypart = entitymodel.PrimaryPart
						local partoffset = part.Position - primarypart.Position
						args[7] = _Vector3new(
							math.random(-originalpart.Size.X, originalpart.Size.X),
							math.random(-originalpart.Size.Y, originalpart.Size.Y),
							math.random(-originalpart.Size.Z, originalpart.Size.Z)
						) + partoffset
					end
				end
			end
		end
		return __gamenamecall(self, unpack(args))
	end
	return __gamenamecall(self, ...)
end)))

do
	local Config = {
		enabled = true, visible = false, max_logs = 8, lifetime = 6.0,
		position = Vector2.new(20, 300), width = 380, line_height = 22,
		font = Drawing.Fonts.System, title = "floppa hub",
		title_color = PURPLE_LIGHT,
		bg_color = Color3.fromRGB(15, 10, 25), bg_alpha = 0.35,
		accent_color = PURPLE,
		kill_color = Color3.fromRGB(255, 90, 90),
		hit_color = Color3.fromRGB(255, 220, 90),
		text_color = Color3.fromRGB(230, 230, 230),
	}
	local Logs, KillCache = {}, {}
	local EntityList = nil
	local function findEntityList()
		for _, v in pairs(getgc(true)) do
			if type(v) == "table" and type(rawget(v, "GetEntityFromPart")) == "function" then
				return rawget(v, "GetEntityFromPart")
			end
		end
	end
	task.spawn(function()
		repeat task.wait(0.5); EntityList = findEntityList() until EntityList
	end)
	local function getEntityByID(id)
		if not EntityList then return nil end
		local ok, list = pcall(function() return getupvalue(EntityList, 1) end)
		if not ok or not list then return nil end
		return list[id]
	end
	local function getEntityName(entity)
		if not entity then return "Unknown" end
		local name = rawget(entity, "Name")
		if name and name ~= "" then return name end
		return rawget(entity, "type") or "Unknown"
	end
	local Drawings = {
		bg = Drawing.new("Square"), bg_outline = Drawing.new("Square"),
		title = Drawing.new("Text"), title_line = Drawing.new("Line"), lines = {},
	}
	Drawings.bg.Filled = true; Drawings.bg.Color = Config.bg_color
	Drawings.bg.Transparency = Config.bg_alpha; Drawings.bg.ZIndex = 90
	Drawings.bg_outline.Filled = false; Drawings.bg_outline.Thickness = 1
	Drawings.bg_outline.Color = Config.accent_color; Drawings.bg_outline.Transparency = 0.7
	Drawings.bg_outline.ZIndex = 91
	Drawings.title.Center = false; Drawings.title.Outline = true
	Drawings.title.OutlineColor = Color3.new(0, 0, 0); Drawings.title.Font = Config.font
	Drawings.title.Size = 14; Drawings.title.Color = Config.title_color
	Drawings.title.Text = Config.title; Drawings.title.ZIndex = 92
	Drawings.title_line.Thickness = 1; Drawings.title_line.Color = Config.accent_color
	Drawings.title_line.Transparency = 0.4; Drawings.title_line.ZIndex = 92
	local function ensureLine(idx)
		if Drawings.lines[idx] then return Drawings.lines[idx] end
		local t = Drawing.new("Text")
		t.Center = false; t.Outline = true; t.OutlineColor = Color3.new(0, 0, 0)
		t.Font = Config.font; t.Size = 13; t.ZIndex = 93
		Drawings.lines[idx] = t
		return t
	end
	local function addLog(text, color)
		table.insert(Logs, 1, { text = text, color = color or Config.text_color, born = tick() })
		while #Logs > Config.max_logs do table.remove(Logs) end
	end
	cheat.utility.new_renderstepped(function()
		local now = tick()
		for i = #Logs, 1, -1 do
			if now - Logs[i].born > Config.lifetime then table.remove(Logs, i) end
		end
		local pos, w = Config.position, Config.width
		if not Config.visible or not Config.enabled or #Logs == 0 then
			Drawings.bg.Visible = false; Drawings.bg_outline.Visible = false
			Drawings.title.Visible = false; Drawings.title_line.Visible = false
			for _, ln in pairs(Drawings.lines) do ln.Visible = false end
			return
		end
		local title_h = 26
		local total_h = title_h + (#Logs * Config.line_height) + 6
		Drawings.bg.Position = pos; Drawings.bg.Size = Vector2.new(w, total_h); Drawings.bg.Visible = true
		Drawings.bg_outline.Position = pos; Drawings.bg_outline.Size = Vector2.new(w, total_h); Drawings.bg_outline.Visible = true
		Drawings.title.Position = pos + Vector2.new(10, 6)
		Drawings.title.Text = Config.title .. "  " .. os.date("%H:%M:%S")
		Drawings.title.Visible = true
		Drawings.title_line.From = pos + Vector2.new(6, title_h - 2)
		Drawings.title_line.To = pos + Vector2.new(w - 6, title_h - 2)
		Drawings.title_line.Visible = true
		for i = 1, Config.max_logs do
			local t = ensureLine(i); local log = Logs[i]
			if log then
				local age = now - log.born
				local fade = mathclamp(1 - (age / Config.lifetime), 0, 1)
				local alpha = math.min(age / 0.2, 1) * fade
				if alpha <= 0.01 then t.Visible = false
				else
					t.Text = string.format("[%s] %s", os.date("%H:%M:%S", math.floor(log.born)), log.text)
					t.Position = pos + Vector2.new(10, title_h + ((i - 1) * Config.line_height) + 2)
					t.Color = log.color; t.Transparency = alpha; t.Visible = true
				end
			else t.Visible = false end
		end
	end)
	UIS.InputBegan:Connect(function(input, gpe)
		if gpe then return end
		if input.KeyCode == Enum.KeyCode.F9 then Config.visible = not Config.visible end
	end)
	local originalNamecall
	originalNamecall = hookmetamethod(game, "__namecall", function(self, ...)
		local method = getnamecallmethod()
		local args = {...}
		if method == "FireServer" and (self == LocalPlayer.TCP or self == LocalPlayer.UDP) then
			if args[1] == 10 and args[2] == "Hit" then
				local entity_id, part_name = args[5], args[6]
				if type(entity_id) == "number" then
					local entity = getEntityByID(entity_id)
					local name = getEntityName(entity)
					local is_head = part_name == "Head"
					local tag = is_head and "HEADSHOT" or "HIT"
					addLog(string.format("> %s %s", tag, name), Config.hit_color)
					task.spawn(function()
						if KillCache[entity_id] then return end
						KillCache[entity_id] = true
						task.wait(0.3)
						local still = getEntityByID(entity_id)
						if still and rawget(still, "sleeping") ~= true then
							task.wait(0.4)
							local still2 = getEntityByID(entity_id)
							if still2 and rawget(still2, "sleeping") ~= true then
								KillCache[entity_id] = nil
								return
							end
						end
						addLog(string.format("+ Killed %s%s", name, is_head and " (HS)" or ""), Config.kill_color)
						KillCache[entity_id] = nil
					end)
				end
			end
		end
		return originalNamecall(self, ...)
	end)
end
